-- ══════════════════════════════════════════════════════════════════
--  XOR (compatible con LuaJ/Lua 5.1 - sin operador ~)
-- ══════════════════════════════════════════════════════════════════

local function bxor(a, b)
    local result = 0
    local bit = 1
    while a > 0 or b > 0 do
        local ra = a % 2
        local rb = b % 2
        if ra ~= rb then result = result + bit end
        a = math.floor(a / 2)
        b = math.floor(b / 2)
        bit = bit * 2
    end
    return result
end

function xor_cifrar(texto, clave)
    local resultado = {}
    for i = 1, #texto do
        local bt = texto:byte(i)
        local bc = clave:byte(((i - 1) % #clave) + 1)
        resultado[i] = string.format("%02X", bxor(bt, bc))
    end
    return table.concat(resultado)
end

function xor_descifrar(hex, clave)
    local resultado = {}
    local j = 1
    for i = 1, #hex - 1, 2 do
        local bh = tonumber(hex:sub(i, i + 1), 16)
        local bc = clave:byte(((j - 1) % #clave) + 1)
        resultado[j] = string.char(bxor(bh, bc))
        j = j + 1
    end
    return table.concat(resultado)
end

-- ══════════════════════════════════════════════════════════════════
--  HELPER DE RED: petición con reintentos automáticos
--  Retorna la respuesta si tuvo éxito, nil si falló todo
-- ══════════════════════════════════════════════════════════════════

function hacer_peticion(url, headers, payload, intentos)
    intentos = intentos or 3
    for i = 1, intentos do
        local ok, res = pcall(function()
            return gg.makeRequest(url, headers, payload)
        end)
        if ok and res and res.content then
            return res
        end
        if i < intentos then gg.sleep(2000) end
    end
    return nil
end

local function asegurar_carpeta(ruta_archivo)
    local carpeta = ruta_archivo:match("^(.+)/[^/]+$")
    if carpeta then
        pcall(function()
            local File = luajava.bindClass("java.io.File")
            local dir = luajava.new(File, carpeta)
            if not dir:exists() then dir:mkdirs() end
        end)
    end
end

--  IDENTIDAD: guardar, leer, borrar, obtener
-- ══════════════════════════════════════════════════════════════════

function guardar_identidad(id, key, exp_ts, created_ts, days)
    local contenido = id .. "|" .. (key or "") .. "|" ..
                      (exp_ts or 0) .. "|" .. (created_ts or 0) .. "|" .. (days or 0)
    local cifrado = xor_cifrar(contenido, CLAVE_XOR)
    local guardados = 0
    for _, ruta in ipairs(RUTAS_SEGURAS) do
        asegurar_carpeta(ruta)
        pcall(function()
            local f = io.open(ruta, "w")
            if f then f:write(cifrado); f:close(); guardados = guardados + 1 end
        end)
    end
    return guardados > 0
end

function leer_identidad()
    for _, ruta in ipairs(RUTAS_SEGURAS) do
        local ok, res = pcall(function()
            local f = io.open(ruta, "r")
            if not f then return nil end
            local contenido = f:read("*a"):gsub("%s+", "")
            f:close()
            if not contenido or #contenido < 16 then return nil end
            local descifrado = xor_descifrar(contenido, CLAVE_XOR)
            local id, key, exp_ts, created_ts, days =
                descifrado:match("([^|]+)|([^|]*)|(%d+)|(%d+)|(%d+)")
            if id and #id >= 8 then
                return { id:upper(), key, tonumber(exp_ts), tonumber(created_ts), tonumber(days) }
            end
            return nil
        end)
        if ok and res then
            return res[1], res[2], res[3], res[4], res[5]
        end
    end
    return nil, nil, nil, nil, nil
end

function borrar_identidad()
    for _, ruta in ipairs(RUTAS_SEGURAS) do
        pcall(function()
            local f = io.open(ruta, "w")
            if f then f:write(""); f:close() end
            os.remove(ruta)
        end)
    end
end

function sincronizar_rutas(id, key, exp_ts, created_ts, days)
    local ok = guardar_identidad(id, key, exp_ts, created_ts, days)
    if ok then
        gg.toast("🔐 Identidad sincronizada en " .. #RUTAS_SEGURAS .. " rutas.")
    end
end

function obtener_solo_id()
    local adn = leer_identidad()
    if adn then return adn end

    local t_now = os.time()
    local gc    = collectgarbage("count")
    local ts_str = tostring(t_now) .. tostring(gc)
    local entropy = 0
    for i = 1, #ts_str do entropy = entropy + ts_str:byte(i) end

    math.randomseed(t_now + entropy * 997 + math.floor(gc * 1000))
    for i = 1, 10 do math.random() end

    local nuevo_id = string.format("%08X%08X%04X",
        math.random(0, 0x7FFFFFFF),
        math.random(0, 0x7FFFFFFF),
        math.random(0, 0xFFFF)
    )
    guardar_identidad(nuevo_id:upper(), "", 0, t_now, 0)
    return nuevo_id:upper()
end

function leer_ultimo_tiempo()
    for _, ruta in ipairs(RUTAS_RELOJ) do
        local ok, res = pcall(function()
            local f = io.open(ruta, "r")
            if not f then return nil end
            local contenido = f:read("*a"):gsub("%s+", "")
            f:close()
            if not contenido or #contenido < 4 then return nil end
            return tonumber(xor_descifrar(contenido, CLAVE_XOR))
        end)
        if ok and res then return res end
    end
    return 0
end

function guardar_ultimo_tiempo(ts)
    local cifrado = xor_cifrar(tostring(math.floor(ts)), CLAVE_XOR)
    for _, ruta in ipairs(RUTAS_RELOJ) do
        asegurar_carpeta(ruta)
        pcall(function()
            local f = io.open(ruta, "w")
            if f then f:write(cifrado); f:close() end
        end)
    end
end

function ejecutar_poder_doble(lista_B, lista_E, nombre)
    if not lista_B or not lista_E then return end
    local res = 0
    gg.setRanges(gg.REGION_ANONYMOUS)
    for i = 1, #lista_B do
        local b_bloque = lista_B[i]
        local e_bloque = lista_E[i]
        -- CORRECCIÓN: nil-check en e_bloque antes de usar #
        if b_bloque and e_bloque and #b_bloque >= POS_4 and #e_bloque >= POS_4 then
            local buscar_hex = b_bloque:sub(POS_1, POS_1+1).." "..b_bloque:sub(POS_2, POS_2+1).." "..b_bloque:sub(POS_3, POS_3+1).." "..b_bloque:sub(POS_4, POS_4+1)
            local editar_hex = e_bloque:sub(POS_1, POS_1+1).." "..e_bloque:sub(POS_2, POS_2+1).." "..e_bloque:sub(POS_3, POS_3+1).." "..e_bloque:sub(POS_4, POS_4+1)
            gg.searchNumber("h"..buscar_hex, gg.TYPE_BYTE)
            local count = gg.getResultCount()
            if count > 0 then
                gg.getResults(count)
                gg.editAll("h"..editar_hex, gg.TYPE_BYTE)
                res = res + count
            end
            gg.clearResults()
        end
    end
    t = t + res
    gg.toast("✅ "..nombre.." finalizado: "..res.." cambios.")
end

function utc_a_unix(anio, mes, dia, hora, min, seg)
    local tm = os.time({
        year=anio, month=mes, day=dia,
        hour=hora, min=min, sec=seg, isdst=false
    })
    local utc_t = os.time(os.date("!*t", tm))
    return tm + (tm - utc_t)
end

function obtener_fecha_real()
    local fuentes = {
        "http://worldtimeapi.org/api/timezone/Etc/UTC",
        "https://timeapi.io/api/Time/current/zone?timeZone=UTC"
    }
    for _, url in ipairs(fuentes) do
        -- CORRECCIÓN: usar hacer_peticion con 2 intentos por fuente
        local solicitud = hacer_peticion(url, nil, nil, 2)
        if solicitud and solicitud.content then
            local unix = solicitud.content:match('"unixtime"%s*:%s*(%d+)')
            if unix then return tonumber(unix) end
            local anio, mes, dia, hora, min, seg = solicitud.content:match(
                '"dateTime"%s*:%s*"(%d+)-(%d+)-(%d+)T(%d+):(%d+):(%d+)')
            if anio then
                return utc_a_unix(tonumber(anio), tonumber(mes), tonumber(dia),
                                  tonumber(hora), tonumber(min), tonumber(seg))
            end
        end
    end
    return nil
end

function validar_tiempo()
    local hora_celular  = os.time()
    local ultimo_valido = leer_ultimo_tiempo()

    -- Anti-retroceso local (funciona sin internet)
    if ultimo_valido > 0 and hora_celular < (ultimo_valido - 60) then
        gg.alert("⚠️ ALERTA: Se detectó un RETROCESO en la fecha/hora de tu dispositivo.\nVuelve a poner la hora correcta para continuar.")
        os.exit()
    end

    local unix_real = obtener_fecha_real()

    -- CORRECCIÓN: Si no hay internet, usar reloj local si hay historial guardado
    if not unix_real then
        if ultimo_valido > 0 then
            gg.toast("⚠️ Sin internet. Usando reloj local.")
            return hora_celular
        else
            gg.alert("❌ Sin conexión y sin historial de tiempo.\nConéctate a internet para continuar.")
            os.exit()
        end
    end

    -- CORRECCIÓN: Tolerancia subida a 10 minutos (600s) para cubrir delays de red
    if math.abs(os.difftime(unix_real, hora_celular)) > 600 then
        gg.alert("⚠️ ALERTA: Fecha/Hora manipulada en los ajustes de tu teléfono.")
        os.exit()
    end

    if unix_real > ultimo_valido then guardar_ultimo_tiempo(unix_real) end
    return unix_real
end

-- CORRECCIÓN: Retorna tabla=éxito | false=key no existe | nil=error de red
function obtener_datos_nube(key_buscada)
    local url = SHEETS_URL .. "?action=buscar_key&key=" .. key_buscada
    local peticion = hacer_peticion(url)

    if not peticion then
        return nil -- Error de red (no hay respuesta tras reintentos)
    end

    if not peticion.content or peticion.content == "[]" or peticion.content == "" then
        return false -- Key no existe en la base de datos
    end

    local datos       = peticion.content
    local db_key      = datos:match('"key"%s*:%s*"([^"]+)"')
    local db_created  = datos:match('"created_at"%s*:%s*(%d+)')
    local db_expires  = datos:match('"expires_at"%s*:%s*(%d+)')
    local db_days     = datos:match('"days_validity"%s*:%s*(%d+)')
    local db_status   = datos:match('"status"%s*:%s*"([^"]+)"')
    local db_device_id = datos:match('"device_id"%s*:%s*"([^"]+)"')

    if not db_key or not db_expires then return false end

    if not db_device_id and datos:match('"device_id"%s*:%s*null') then
        db_device_id = "null"
    end

    return {
        key           = db_key,
        created_at    = tonumber(db_created),
        expires_at    = tonumber(db_expires),
        days_validity = tonumber(db_days),
        status        = db_status,
        device_id     = db_device_id
    }
end

function vincular_key_supabase(key_buscada, mi_adn)
    local payload = string.format(
        '{"action": "vincular_key", "key_buscada": "%s", "mi_adn": "%s"}',
        key_buscada, mi_adn
    )
    local headers = {["Content-Type"] = "application/json"}
    local peticion = hacer_peticion(SHEETS_URL, headers, payload)

    if peticion then
        if peticion.code == 200 then
            if peticion.content and peticion.content:match('"status"%s*:%s*"success"') then
                return true
            else
                local msg = "Error desconocido"
                if peticion.content then
                    msg = peticion.content:match('"message"%s*:%s*"([^"]+)"') or msg
                end
                gg.alert("❌ Falla al vincular key: " .. msg)
                return false
            end
        else
            gg.alert("❌ Falla en la nube. Código: " .. tostring(peticion.code))
            return false
        end
    end

    gg.alert("❌ El teléfono no tiene internet o hay un problema de conexión.")
    return false
end

                    borrar_identidad()

                                      nube.created_at, nube.days_validity)

        gg.alert("🚫 ESTA KEY ESTÁ PAUSADA O DESACTIVADA.")
        return
    end

    local tiempo_restante = nube.expires_at - unix_real

                          nube.created_at, nube.days_validity)

        os.exit()
    end

                        local exp_str = string.format("%02d%02d%04d",
                            fecha_exp.day, fecha_exp.month, fecha_exp.year)
                        local k = random_prefix .. "_EXP_" .. exp_str

                        local ahora_ts = os.time()
                        local payload = string.format(
                            '{"action":"crear_key","master_pass":"%s","key":"%s","created_at":%d,"expires_at":%d,"days_validity":%d}',
                            MASTER_PASS, k, ahora_ts, tiempo_exp, dias_validez
                        )
                        local headers = {["Content-Type"] = "application/json"}
                        -- CORRECCIÓN: usar hacer_peticion y verificar respuesta
                        local resp = hacer_peticion(SHEETS_URL, headers, payload)
                        if resp and resp.content and resp.content:match('"status"%s*:%s*"success"') then
                            gg.alert("✅ KEY CREADA:\n\n" .. k ..
                                     "\n\nVence: " .. fecha_exp.day .. "/" ..
                                     fecha_exp.month .. "/" .. fecha_exp.year)
                        else
                            gg.alert("⚠️ KEY GENERADA (verifica en tu hoja):\n\n" .. k)
                        end
                    end

                -- ── RESETEAR DISPOSITIVO ───────────────────────────
                elseif accion_admin == 2 then
                    local p_key = gg.prompt({"🔑 KEY a desvincular:"}, {""}, {"text"})
                    if p_key and p_key[1] ~= "" then
                        local key_reset = p_key[1]:gsub("%s+", ""):upper()
                        gg.toast("⏳ Desvinculando dispositivo...")
                        local payload = string.format(
                            '{"action":"resetear_dispositivo","master_pass":"%s","key":"%s"}',
                            MASTER_PASS, key_reset
                        )
                        local headers = {["Content-Type"] = "application/json"}
                        local resp = hacer_peticion(SHEETS_URL, headers, payload)
                        if resp and resp.content then
                            if resp.content:match('"status"%s*:%s*"success"') then
                                gg.alert("✅ DISPOSITIVO DESVINCULADO.\nLa key puede usarse en un dispositivo nuevo.")
                            else
                                local msg = resp.content:match('"message"%s*:%s*"([^"]+)"') or "Error desconocido"
                                gg.alert("❌ Error: " .. msg)
                            end
                        else
                            gg.alert("❌ Sin respuesta del servidor.")
                        end
                    end

                -- ── ELIMINAR KEY ───────────────────────────────────
                elseif accion_admin == 3 then
                    local p_key = gg.prompt({"🗑️ KEY a eliminar:"}, {""}, {"text"})
                    if p_key and p_key[1] ~= "" then
                        local key_del = p_key[1]:gsub("%s+", ""):upper()
                        local conf = gg.choice({"Sí, eliminarla", "No, cancelar"}, nil,
                            "¿Eliminar la key " .. key_del .. "?")
                        if conf == 1 then
                            local payload = string.format(
                                '{"action":"delete","master_pass":"%s","key":"%s"}',
                                MASTER_PASS, key_del
                            )
                            local headers = {["Content-Type"] = "application/json"}
                            local resp = hacer_peticion(SHEETS_URL, headers, payload)
                            if resp and resp.content and resp.content:match('"status"%s*:%s*"success"') then
                                gg.alert("✅ KEY ELIMINADA.")
                            else
                                gg.alert("❌ No se pudo eliminar la key.")
                            end
                        end
                    end
                end
            end

        elseif opt == 3 then
            os.exit()
        end
    end
    gg.sleep(100)
end
